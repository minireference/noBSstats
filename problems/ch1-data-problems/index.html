P1.15¶

Your friend Ben has written the following function for generating a random sample of size n from a given population.

import numpy as np
from numpy.random import choice

def get_sample(population, n):
    N = len(population)
    nmore, nless = N//2, N - N//2
    ws = np.concat([5*np.ones(nmore), np.ones(nless)])
    p = ws / np.sum(ws)
    sample = choice(population, size=n, replace=False, p=p)
    return sample
